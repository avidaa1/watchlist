function decryptBlob(blob,passphrase,onURL){

  const combined=new Uint8Array(data.length+tag.length);combined.set(data);combined.set(tag,data.length);
  const enc=new TextEncoder();
  return crypto.subtle.importKey('raw',enc.encode(passphrase),{name:'PBKDF2'},false,['deriveKey'])

/* ===== AUTO MEDIA LOOKUP — reads the page's MEDIA[] if present (DRY, auto-updates) ===== */
function mediaMatch(low){
  const M = (typeof window!=='undefined' && Array.isArray(window.MEDIA)) ? window.MEDIA : (typeof MEDIA!=='undefined'?MEDIA:null);
  if(!M||!M.length) return null;
  /* catalogue-level questions */
  if(/\b(list|all|how many|what|which)\b/.test(low) && /\b(film|films|movie|movies|book|books|show|shows|anime|title|titles|media|library|archive)\b/.test(low)){
    const sorted=M.slice().sort((a,b)=>b.year-a.year);
    const names=sorted.map(m=>m.title+' ('+m.year+')').join(' · ');
    return {text:'This archive holds '+M.length+' entries — newest first:\n'+names,link:null};
  }
  /* specific title match: longest title whose words appear in the query */
  let hit=null,hitLen=0;
  M.forEach(m=>{
    const t=m.title.toLowerCase();
    const core=t.replace(/[^a-z0-9 ]/g,'').trim();
    if(core.length>=3 && (low.includes(core)||low.includes(core.split(':')[0].trim())) && core.length>hitLen){hit=m;hitLen=core.length;}
  });
  if(hit){
    return {text:hit.title+' ('+hit.year+'), '+hit.creator+'. '+hit.note,link:null};
  }
  return null;
}

  for(let j=0;j<=n;j++)d[0][j]=j;
  for(let i=1;i<=m;i++)for(let j=1;j<=n;j++)
    d[i][j]=Math.min(d[i-1][j]+1,d[i][j-1]+1,d[i-1][j-1]+(a[i-1]===b[j-1]?0:1));
  return d[m][n];
}

/* detect the user's register so we can mirror it */
function detectTone(msg){
  const m=msg.toLowerCase();
  if(/[!]{1,}|wow|amazing|cool|awesome|love|🔥|insane|epic/.test(m))return'excited';
  if(/\?$|how|why|what|explain|architecture|works|technical|api|code|spec/.test(m))return'curious';
  if(/really|prove|sure|doubt|fake|scam|nonsense|why should/.test(m))return'skeptical';
  if(/\b(hey|sup|yo|lol|wassup|gimme|u |ur )\b/.test(m))return'casual';
  return'neutral';
}

/* score one KB entry against the message, with fuzzy/typo tolerance */
function scoreEntry(e,low,words){
  let sc=0;
  e.k.forEach(k=>{
    if(low===k||words.includes(k))sc+=10;                /* exact keyword/word match — always confident */
    else if(low.includes(k)&&k.length>=4)sc+=k.split(' ').length*k.length*2;  /* substring, but only for keywords 4+ */
    else if(k.startsWith(low)||low.startsWith(k))sc+=Math.min(k.length,low.length)*1.2;
    else words.forEach(w=>{
      if(w.length>=3&&k.startsWith(w))sc+=w.length*0.9;
      else if(w.length>=4&&k.includes(w))sc+=w.length*0.5;
      else if(w.length>=6&&k.length>=6&&lev(w,k)<=1)sc+=w.length*0.6; /* typo tolerance, long words only */
    });
  });
  return sc;
}

/* build a composed, adaptive reply from one or two KB hits */
function compose(hits,tone){
  /* pick a fresh variant from the top hit (no-repeat) */
  const top=hits[0].e, v=top.r;
  let i=Math.floor(Math.random()*v.length);
  if(v.length>1&&chatLastResponse===v[i])i=(i+1)%v.length;
  let body=v[i];
  /* memory: if we've talked before, occasionally tie back */
  if(chatSession.turns>2 && chatSession.topics.length>1 && Math.random()<0.28){
    const prev=chatSession.topics[chatSession.topics.length-1];
    if(prev && prev!==top._id){
      const bridges=[' This ties into '+prev+', which you asked about.',' Like '+prev+', it\u2019s part of the same signal.',''];
      body+=bridges[Math.floor(Math.random()*bridges.length)];
    }
  }
  /* multi-topic blend: weave a second strong hit */
  if(hits[1] && hits[1].sc>=hits[0].sc*0.6 && hits[1].e!==top){
    const v2=hits[1].e.r; const seg=v2[Math.floor(Math.random()*v2.length)];
    body+=' '+seg;
  }
  chatLastResponse=v[i];
  top._id=top._id||top.k[0];
  chatSession.topics.push(top._id);
  return {text:flavor(body,tone),link:top.link||null};
}

function chatMatch(msg){
  const low=msg.toLowerCase().trim();
  chatSession.turns++; chatSession.asked.push(low);
  const tone=detectTone(msg); chatSession.lastTone=tone;

  const quest=questCheck(low);
  if(quest)return quest; /* quests stay literal — no flavoring; {handled:true} skips rendering */

  /* frustration: acknowledge & back off instead of repeating the pitch */
  if(isFrustrated(low))return {text:BACKOFF[Math.floor(Math.random()*BACKOFF.length)],link:null};

  const words=low.split(/\s+/).filter(w=>w.length>=2);

  /* smalltalk first — minimal redirect, no forced topic */
  const st=smalltalkMatch(low,words);
  if(st)return {text:st,link:null};

  /* rank ALL KB entries (not just best) so we can blend */
  const ranked=KB.filter(e=>!(e.r[0]&&e.r[0].startsWith('__QUEST')))
    .map(e=>({e,sc:scoreEntry(e,low,words)}))
    .filter(x=>x.sc>0)
    .sort((a,b)=>b.sc-a.sc);

  const media=mediaMatch(low);
  const bestScore=ranked.length?ranked[0].sc:0;

  /* film hit wins when KB is weak */
  if(media && bestScore<8){chatLastResponse=media.text;return {text:flavor(media.text,tone),link:media.link||null};}
  /* KB answer only if confidently above the noise floor */
  if(ranked.length && bestScore>=6)return compose(ranked,tone);
  if(media){chatLastResponse=media.text;return {text:flavor(media.text,tone),link:media.link||null};}

  /* playful but honest redirect — no random page */
  return {text:REDIRECTS[Math.floor(Math.random()*REDIRECTS.length)],link:null};
}
